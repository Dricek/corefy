// Fonction pour parser les quantités
function parseQuantity(quantityStr) {
  if (!quantityStr) return 100;
  
  const qty = quantityStr.toLowerCase().trim();
  const numbers = qty.match(/\d+(?:\.\d+)?/);
  if (!numbers) return 100;
  
  const amount = parseFloat(numbers[0]);
  
  if (qty.includes('g')) return amount;
  if (qty.includes('kg')) return amount * 1000;
  if (qty.includes('ml')) return amount; // Approximation 1ml ≈ 1g
  if (qty.includes('cuillère') || qty.includes('càs')) return amount * 15;
  if (qty.includes('càc')) return amount * 5;
  if (qty.includes('tasse')) return amount * 150;
  
  return amount;
}

// Fonction pour trouver l'ingrédient dans la DB
function findNutrition(ingredient) {
  const ing = ingredient.toLowerCase();
  
  // Recherche exacte
  if (nutritionDB[ing]) return nutritionDB[ing];
  
  // Recherche par mots-clés
  for (const [dbIng, nutrition] of Object.entries(nutritionDB)) {
    if (dbIng.split(' ').some(word => ing.includes(word))) {
      return nutrition;
    }
  }
  
  // Valeur par défaut (légume générique)
  return {"kcal": 25, "proteines": 2, "glucides": 4, "lipides": 0.2};
}

// Fonction pour calculer les macros d'une recette
function calculateRecipeMacros(ingredients) {
  const total = {"kcal": 0, "proteines": 0, "glucides": 0, "lipides": 0};
  
  ingredients.forEach(([ingredient, quantity]) => {
    const nutrition = findNutrition(ingredient);
    const grams = parseQuantity(quantity);
    const multiplier = grams / 100;
    
    Object.keys(total).forEach(macro => {
      total[macro] += nutrition[macro] * multiplier;
    });
  });
  
  // Arrondir
  Object.keys(total).forEach(key => {
    total[key] = Math.round(total[key] * 10) / 10;
  });
  
  return total;
}

// Fonction pour générer l'affichage des macros
function generateMacrosHTML(macros) {
  return `
    <div class="recipe-macros">
      <span class="kcal">${macros.kcal} kcal</span>
      <span class="macro protein">🥩 ${macros.proteines}g</span>
      <span class="macro carbs">🍞 ${macros.glucides}g</span>
      <span class="macro fats">🥑 ${macros.lipides}g</span>
    </div>
  `;
}
